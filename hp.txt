#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

typedef enum {
    TOK_NUM,
    TOK_PLUS,
    TOK_MINUS,
    TOK_MUL,
    TOK_DIV,
    TOK_LPAREN,
    TOK_RPAREN,
    TOK_END,
    TOK_INVALID
} TokenType;

typedef struct {
    TokenType type;
    char text[128];
} Token;

static Token tokens[1024];
static int token_count = 0;
static int cur_token = 0;

/* Check if string is a valid scientific or float or int constant */
static int scan_number(const char *src, int *idx, char *out) {
    int i = *idx;
    int start = i;
    int has_dot = 0;
    int has_exp = 0;

    // Digits before dot
    if (!isdigit((unsigned char)src[i])) return 0;
    while (isdigit((unsigned char)src[i])) i++;

    // Optional fractional part
    if (src[i] == '.') {
        // Lookahead: must have digit after dot
        if (!isdigit((unsigned char)src[i + 1])) {
            return 0; // Invalid like 12.
        }
        has_dot = 1;
        i++;
        while (isdigit((unsigned char)src[i])) i++;
    }

    // Optional exponent part
    if (src[i] == 'e' || src[i] == 'E') {
        int e_idx = i;
        i++;
        if (src[i] == '+' || src[i] == '-') i++;
        if (!isdigit((unsigned char)src[i])) {
            return 0; // Invalid like 2e
        }
        while (isdigit((unsigned char)src[i])) i++;
        has_exp = 1;
    }

    // Check if next char is dot (e.g. 3.4.5)
    if (src[i] == '.') return 0;

    int len = i - start;
    strncpy(out, src + start, len);
    out[len] = '\0';
    *idx = i;
    return 1;
}

static int tokenize(const char *input) {
    token_count = 0;
    cur_token = 0;
    int i = 0;

    while (input[i] != '\0') {
        if (isspace((unsigned char)input[i])) {
            i++;
            continue;
        }

        if (isdigit((unsigned char)input[i])) {
            char num_buf[128];
            if (scan_number(input, &i, num_buf)) {
                tokens[token_count].type = TOK_NUM;
                strcpy(tokens[token_count].text, num_buf);
                token_count++;
            } else {
                tokens[token_count].type = TOK_INVALID;
                token_count++;
                return 0;
            }
        } else if (input[i] == '+') {
            tokens[token_count].type = TOK_PLUS;
            strcpy(tokens[token_count].text, "+");
            token_count++;
            i++;
        } else if (input[i] == '-') {
            tokens[token_count].type = TOK_MINUS;
            strcpy(tokens[token_count].text, "-");
            token_count++;
            i++;
        } else if (input[i] == '*') {
            tokens[token_count].type = TOK_MUL;
            strcpy(tokens[token_count].text, "*");
            token_count++;
            i++;
        } else if (input[i] == '/') {
            tokens[token_count].type = TOK_DIV;
            strcpy(tokens[token_count].text, "/");
            token_count++;
            i++;
        } else if (input[i] == '(') {
            tokens[token_count].type = TOK_LPAREN;
            strcpy(tokens[token_count].text, "(");
            token_count++;
            i++;
        } else if (input[i] == ')') {
            tokens[token_count].type = TOK_RPAREN;
            strcpy(tokens[token_count].text, ")");
            token_count++;
            i++;
        } else {
            tokens[token_count].type = TOK_INVALID;
            token_count++;
            return 0;
        }
    }

    tokens[token_count].type = TOK_END;
    tokens[token_count].text[0] = '\0';
    token_count++;
    return 1;
}

/* Recursive Descent Parser adhering to Yacc precedence */
static int parse_E();
static int parse_T();
static int parse_F();

static int parse_F() {
    if (tokens[cur_token].type == TOK_NUM) {
        cur_token++;
        return 1;
    } else if (tokens[cur_token].type == TOK_LPAREN) {
        cur_token++;
        if (!parse_E()) return 0;
        if (tokens[cur_token].type == TOK_RPAREN) {
            cur_token++;
            return 1;
        }
        return 0;
    }
    return 0;
}

static int parse_T() {
    if (!parse_F()) return 0;
    while (tokens[cur_token].type == TOK_MUL || tokens[cur_token].type == TOK_DIV) {
        cur_token++;
        if (!parse_F()) return 0;
    }
    return 1;
}

static int parse_E() {
    if (!parse_T()) return 0;
    while (tokens[cur_token].type == TOK_PLUS || tokens[cur_token].type == TOK_MINUS) {
        cur_token++;
        if (!parse_T()) return 0;
    }
    return 1;
}

int validate_arithmetic_expression(const char *expr) {
    if (!tokenize(expr)) return 0;
    if (tokens[0].type == TOK_END) return 0; // empty expression
    if (!parse_E()) return 0;
    if (tokens[cur_token].type != TOK_END) return 0;
    return 1;
}

int main(int argc, char *argv[]) {
    char input[256];

    printf("Enter expression:\n");
    if (fgets(input, sizeof(input), stdin)) {
        // remove trailing newline
        size_t len = strlen(input);
        if (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {
            input[len - 1] = '\0';
        }
        if (len > 1 && (input[len - 2] == '\r')) {
            input[len - 2] = '\0';
        }

        if (validate_arithmetic_expression(input)) {
            printf("Valid arithmetic expression\n");
        } else {
            printf("Invalid arithmetic expression\n");
        }
    }
    return 0;
}

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

typedef enum {
    TOK_ID,
    TOK_NUM,
    TOK_ASSIGN,
    TOK_PLUS,
    TOK_MINUS,
    TOK_MUL,
    TOK_DIV,
    TOK_LPAREN,
    TOK_RPAREN,
    TOK_SEMICOLON,
    TOK_END,
    TOK_INVALID
} TokenType;

typedef struct {
    TokenType type;
    char text[64];
} Token;

static Token tokens[4096];
static int token_count = 0;
static int cur_token = 0;

static int scan_number(const char *src, int *idx, char *out) {
    int i = *idx;
    int start = i;

    if (!isdigit((unsigned char)src[i])) return 0;
    while (isdigit((unsigned char)src[i])) i++;

    if (src[i] == '.') {
        if (!isdigit((unsigned char)src[i + 1])) return 0;
        i++;
        while (isdigit((unsigned char)src[i])) i++;
    }

    if (src[i] == '.') return 0;

    int len = i - start;
    strncpy(out, src + start, len);
    out[len] = '\0';
    *idx = i;
    return 1;
}

static int tokenize(const char *input) {
    token_count = 0;
    cur_token = 0;
    int i = 0;

    while (input[i] != '\0') {
        if (isspace((unsigned char)input[i])) {
            i++;
            continue;
        }

        if (isalpha((unsigned char)input[i]) || input[i] == '_') {
            int start = i;
            while (isalnum((unsigned char)input[i]) || input[i] == '_') i++;
            int len = i - start;
            tokens[token_count].type = TOK_ID;
            strncpy(tokens[token_count].text, input + start, len);
            tokens[token_count].text[len] = '\0';
            token_count++;
        } else if (isdigit((unsigned char)input[i])) {
            char num_buf[64];
            if (scan_number(input, &i, num_buf)) {
                tokens[token_count].type = TOK_NUM;
                strcpy(tokens[token_count].text, num_buf);
                token_count++;
            } else {
                tokens[token_count].type = TOK_INVALID;
                token_count++;
                return 0;
            }
        } else if (input[i] == '=') {
            tokens[token_count].type = TOK_ASSIGN;
            strcpy(tokens[token_count].text, "=");
            token_count++;
            i++;
        } else if (input[i] == ';') {
            tokens[token_count].type = TOK_SEMICOLON;
            strcpy(tokens[token_count].text, ";");
            token_count++;
            i++;
        } else if (input[i] == '+') {
            tokens[token_count].type = TOK_PLUS;
            strcpy(tokens[token_count].text, "+");
            token_count++;
            i++;
        } else if (input[i] == '-') {
            tokens[token_count].type = TOK_MINUS;
            strcpy(tokens[token_count].text, "-");
            token_count++;
            i++;
        } else if (input[i] == '*') {
            tokens[token_count].type = TOK_MUL;
            strcpy(tokens[token_count].text, "*");
            token_count++;
            i++;
        } else if (input[i] == '/') {
            tokens[token_count].type = TOK_DIV;
            strcpy(tokens[token_count].text, "/");
            token_count++;
            i++;
        } else if (input[i] == '(') {
            tokens[token_count].type = TOK_LPAREN;
            strcpy(tokens[token_count].text, "(");
            token_count++;
            i++;
        } else if (input[i] == ')') {
            tokens[token_count].type = TOK_RPAREN;
            strcpy(tokens[token_count].text, ")");
            token_count++;
            i++;
        } else {
            tokens[token_count].type = TOK_INVALID;
            token_count++;
            return 0;
        }
    }

    tokens[token_count].type = TOK_END;
    tokens[token_count].text[0] = '\0';
    token_count++;
    return 1;
}

static int parse_E(void);
static int parse_T(void);
static int parse_F(void);
static int parse_statement(void);
static int parse_statement_list(void);

static int parse_F(void) {
    if (tokens[cur_token].type == TOK_ID || tokens[cur_token].type == TOK_NUM) {
        cur_token++;
        return 1;
    }
    if (tokens[cur_token].type == TOK_LPAREN) {
        cur_token++;
        if (!parse_E()) return 0;
        if (tokens[cur_token].type != TOK_RPAREN) return 0;
        cur_token++;
        return 1;
    }
    return 0;
}

static int parse_T(void) {
    if (!parse_F()) return 0;
    while (tokens[cur_token].type == TOK_MUL || tokens[cur_token].type == TOK_DIV) {
        cur_token++;
        if (!parse_F()) return 0;
    }
    return 1;
}

static int parse_E(void) {
    if (!parse_T()) return 0;
    while (tokens[cur_token].type == TOK_PLUS || tokens[cur_token].type == TOK_MINUS) {
        cur_token++;
        if (!parse_T()) return 0;
    }
    return 1;
}

static int parse_statement(void) {
    if (tokens[cur_token].type != TOK_ID) return 0;
    cur_token++;
    if (tokens[cur_token].type != TOK_ASSIGN) return 0;
    cur_token++;
    if (!parse_E()) return 0;
    if (tokens[cur_token].type != TOK_SEMICOLON) return 0;
    cur_token++;
    return 1;
}

static int parse_statement_list(void) {
    if (!parse_statement()) return 0;
    while (tokens[cur_token].type != TOK_END) {
        if (!parse_statement()) return 0;
    }
    return 1;
}

int validate_assignment_sequence(const char *input) {
    if (!tokenize(input)) return 0;
    if (tokens[0].type == TOK_END) return 0; // Empty input is not a valid statement list
    if (!parse_statement_list()) return 0;
    if (tokens[cur_token].type != TOK_END) return 0;
    return 1;
}

int main(void) {
    char buffer[16384];
    size_t total = 0;
    char line[512];

    printf("Enter sequence of assignment statements (Press EOF or blank line on terminal to end):\n");
    buffer[0] = '\0';

    while (fgets(line, sizeof(line), stdin)) {
        size_t len = strlen(line);
        if (len + total < sizeof(buffer) - 1) {
            strcat(buffer, line);
            total += len;
        }
    }

    if (validate_assignment_sequence(buffer)) {
        printf("Valid sequence of assignment statements\n");
    } else {

typedef enum {
    TOK_ID,
    TOK_NUM,
    TOK_ASSIGN,
    TOK_PLUS,
    TOK_MINUS,
    TOK_MUL,
    TOK_DIV,
    TOK_LPAREN,

    TOK_SEMICOLON,
    TOK_END,
    TOK_INVALID
} TokenType;

typedef struct {
    TokenType type;
    char text[64];
} Token;

static Token tokens[1024];
static int token_count = 0;
static int cur_token = 0;

static int scan_number(const char *src, int *idx, char *out) {
    int i = *idx;
    int start = i;

    if (!isdigit((unsigned char)src[i])) return 0;
    while (isdigit((unsigned char)src[i])) i++;

    if (src[i] == '.') {
        if (!isdigit((unsigned char)src[i + 1])) return 0; // invalid e.g. 10.
        i++;
        while (isdigit((unsigned char)src[i])) i++;
    }

    if (src[i] == '.') return 0; // extra dot

    int len = i - start;
    strncpy(out, src + start, len);
    out[len] = '\0';
    *idx = i;
    return 1;
}

static int tokenize(const char *input) {
    token_count = 0;
    cur_token = 0;
    int i = 0;

    while (input[i] != '\0') {
        if (isspace((unsigned char)input[i])) {
            i++;
            continue;
        }

        if (isalpha((unsigned char)input[i]) || input[i] == '_') {
            int start = i;
            while (isalnum((unsigned char)input[i]) || input[i] == '_') i++;
            int len = i - start;
            tokens[token_count].type = TOK_ID;
            strncpy(tokens[token_count].text, input + start, len);
            tokens[token_count].text[len] = '\0';
            token_count++;
        } else if (isdigit((unsigned char)input[i])) {
            char num_buf[64];
            if (scan_number(input, &i, num_buf)) {
                tokens[token_count].type = TOK_NUM;
                strcpy(tokens[token_count].text, num_buf);
                token_count++;
            } else {
                tokens[token_count].type = TOK_INVALID;
                token_count++;
                return 0;
            }
        } else if (input[i] == '=') {
            tokens[token_count].type = TOK_ASSIGN;
            strcpy(tokens[token_count].text, "=");
            token_count++;
            i++;
        } else if (input[i] == ';') {
            tokens[token_count].type = TOK_SEMICOLON;
            strcpy(tokens[token_count].text, ";");
            token_count++;
            i++;
        } else if (input[i] == '+') {
            tokens[token_count].type = TOK_PLUS;
            strcpy(tokens[token_count].text, "+");
            token_count++;
            i++;
        } else if (input[i] == '-') {
            tokens[token_count].type = TOK_MINUS;
            strcpy(tokens[token_count].text, "-");
            token_count++;
            i++;
        } else if (input[i] == '*') {
            tokens[token_count].type = TOK_MUL;
            strcpy(tokens[token_count].text, "*");
            token_count++;
            i++;
        } else if (input[i] == '/') {
            tokens[token_count].type = TOK_DIV;
            strcpy(tokens[token_count].text, "/");
            token_count++;
            i++;
        } else if (input[i] == '(') {
            tokens[token_count].type = TOK_LPAREN;
            strcpy(tokens[token_count].text, "(");
            token_count++;
            i++;
        } else if (input[i] == ')') {
            tokens[token_count].type = TOK_RPAREN;
            strcpy(tokens[token_count].text, ")");
            token_count++;
            i++;
        } else {
            tokens[token_count].type = TOK_INVALID;
            token_count++;
            return 0;
        }
    }

    tokens[token_count].type = TOK_END;
    tokens[token_count].text[0] = '\0';
    token_count++;
    return 1;
}

static int parse_E(void);
static int parse_T(void);
static int parse_F(void);

static int parse_F(void) {
    if (tokens[cur_token].type == TOK_ID || tokens[cur_token].type == TOK_NUM) {
        cur_token++;
        return 1;
    }
    if (tokens[cur_token].type == TOK_LPAREN) {
        cur_token++;
        if (!parse_E()) return 0;
        if (tokens[cur_token].type != TOK_RPAREN) return 0;
        cur_token++;
        return 1;
    }
    return 0;
}

static int parse_T(void) {
    if (!parse_F()) return 0;
    while (tokens[cur_token].type == TOK_MUL || tokens[cur_token].type == TOK_DIV) {
        cur_token++;
        if (!parse_F()) return 0;
    }
    return 1;
}

static int parse_E(void) {
    if (!parse_T()) return 0;
    while (tokens[cur_token].type == TOK_PLUS || tokens[cur_token].type == TOK_MINUS) {
        cur_token++;
        if (!parse_T()) return 0;
    }
    return 1;
}

int validate_assignment_statement(const char *stmt) {

    if (tokens[cur_token].type != TOK_ID) return 0;
    cur_token++;
    if (tokens[cur_token].type != TOK_ASSIGN) return 0;
    cur_token++;
    if (!parse_E()) return 0;
    if (tokens[cur_token].type != TOK_SEMICOLON) return 0;
    cur_token++;
    if (tokens[cur_token].type != TOK_END) return 0;
    return 1;
}

#include <stddef.h>

#include <string.h>
#include <ctype.h>

void lex_sequence_stream(void) {
    int ch;
    while ((ch = getchar()) != EOF) {
        if (isspace(ch)) {
            continue;
        }

        if (isalpha(ch) || ch == '_') {
            char id_buf[128];
            int len = 0;
            id_buf[len++] = (char)ch;
            while ((ch = getchar()) != EOF && (isalnum(ch) || ch == '_')) {
                if (len < 127) id_buf[len++] = (char)ch;
            }
            id_buf[len] = '\0';
            printf("IDENTIFIER: %s\n", id_buf);
            if (ch != EOF) ungetc(ch, stdin);
        } else if (isdigit(ch)) {
            char num_buf[64];
            int len = 0;
            int is_float = 0;
            num_buf[len++] = (char)ch;
            while ((ch = getchar()) != EOF && isdigit(ch)) {
                if (len < 63) num_buf[len++] = (char)ch;
            }
            if (ch == '.') {
                int next = getchar();
                if (isdigit(next)) {
                    is_float = 1;
                    if (len < 63) num_buf[len++] = '.';
                    if (len < 63) num_buf[len++] = (char)next;
                    while ((ch = getchar()) != EOF && isdigit(ch)) {
                        if (len < 63) num_buf[len++] = (char)ch;
                    }
                    if (ch != EOF) ungetc(ch, stdin);
                } else {
                    if (next != EOF) ungetc(next, stdin);
                    ungetc('.', stdin);
                }
            } else if (ch != EOF) {
                ungetc(ch, stdin);
            }
            num_buf[len] = '\0';
            if (is_float) {
                printf("FLOAT: %s\n", num_buf);
            } else {
                printf("INTEGER: %s\n", num_buf);
            }
        } else if (ch == '=') {
            printf("ASSIGN: =\n");
        } else if (ch == ';') {
            printf("SEMICOLON: ;\n");
        } else if (ch == '+') {
            printf("PLUS: +\n");
        } else if (ch == '-') {
            printf("MINUS: -\n");
        } else if (ch == '*') {
            printf("MULTIPLY: *\n");
        } else if (ch == '/') {
            printf("DIVIDE: /\n");
        } else if (ch == '(') {
            printf("LPAREN: (\n");
        } else if (ch == ')') {
            printf("RPAREN: )\n");
        } else {
            printf("UNKNOWN: %c\n", ch);
        }
    }
}

int main(void) {
    lex_sequence_stream();
    return 0;
}

void lex_assignment_statement(const char *input) {
    int i = 0;

    while (input[i] != '\0') {
        if (isspace((unsigned char)input[i])) {
            i++;
            continue;
        }

        if (isalpha((unsigned char)input[i]) || input[i] == '_') {
            int start = i;
            while (isalnum((unsigned char)input[i]) || input[i] == '_') i++;
            int len = i - start;
            char id_buf[128];
            strncpy(id_buf, input + start, len);
            id_buf[len] = '\0';
            printf("IDENTIFIER: %s\n", id_buf);
        } else if (isdigit((unsigned char)input[i])) {
            int start = i;
            int is_float = 0;
            while (isdigit((unsigned char)input[i])) i++;
            if (input[i] == '.' && isdigit((unsigned char)input[i + 1])) {
                is_float = 1;
                i++;
                while (isdigit((unsigned char)input[i])) i++;
            }
            int len = i - start;
            char num_buf[64];
            strncpy(num_buf, input + start, len);
            num_buf[len] = '\0';
            if (is_float) {
                printf("FLOAT: %s\n", num_buf);
            } else {
                printf("INTEGER: %s\n", num_buf);
            }
        } else if (input[i] == '=') {
            printf("ASSIGN: =\n");
            i++;
        } else if (input[i] == ';') {
            printf("SEMICOLON: ;\n");
            i++;
        } else if (input[i] == '+') {
            printf("PLUS: +\n");
            i++;
        } else if (input[i] == '-') {
            printf("MINUS: -\n");
            i++;
        } else if (input[i] == '*') {
            printf("MULTIPLY: *\n");
            i++;
        } else if (input[i] == '/') {
            printf("DIVIDE: /\n");
            i++;
        } else if (input[i] == '(') {
            printf("LPAREN: (\n");
            i++;
        } else if (input[i] == ')') {
            printf("RPAREN: )\n");
            i++;
        } else {
            printf("UNKNOWN: %c\n", input[i]);
            i++;
        }
    }
}

int main(void) {
    char input[1024];

/*
Returns:
  1: Scientific
  2: Float
  3: Integer
  0: Invalid numeric constant (e.g. 12., .25, 2e, 3.4.5)
*/
static int classify_number(const char *src, int *idx, char *token_str, int *out_type) {
    int i = *idx;
    int start = i;

    // Must start with a digit (rejects .25)
    if (!isdigit((unsigned char)src[i])) return 0;
    while (isdigit((unsigned char)src[i])) i++;

    int is_float = 0;
    int is_sci = 0;

    // Optional fractional part
    if (src[i] == '.') {
        // Lookahead: digit must follow '.'
        if (!isdigit((unsigned char)src[i + 1])) {
            // e.g. 12. - invalid constant
            int len = (i + 1) - start;
            strncpy(token_str, src + start, len);
            token_str[len] = '\0';
            *idx = i + 1;
            return 0;
        }
        is_float = 1;
        i++;
        while (isdigit((unsigned char)src[i])) i++;
    }

    // Optional exponent part
    if (src[i] == 'e' || src[i] == 'E') {
        int e_idx = i;
        int next_idx = i + 1;
        if (src[next_idx] == '+' || src[next_idx] == '-') next_idx++;
        if (!isdigit((unsigned char)src[next_idx])) {
            // e.g. 2e - invalid constant
            int len = (e_idx + 1) - start;
            strncpy(token_str, src + start, len);
            token_str[len] = '\0';
            *idx = e_idx + 1;
            return 0;
        }
        is_sci = 1;
        i = next_idx;
        while (isdigit((unsigned char)src[i])) i++;
    }

    // Check if there is another '.' immediately following (e.g. 3.4.5)
    if (src[i] == '.') {
        // Malformed numeric string like 3.4.5
        while (isdigit((unsigned char)src[i]) || src[i] == '.') i++;
        int len = i - start;
        strncpy(token_str, src + start, len);
        token_str[len] = '\0';
        *idx = i;
        return 0;
    }

    int len = i - start;
    strncpy(token_str, src + start, len);
    token_str[len] = '\0';
    *idx = i;

    if (is_sci) {
        *out_type = 1;
    } else if (is_float) {
        *out_type = 2;
    } else {
        *out_type = 3;
    }
    return 1;
}

void lex_numeric_and_arithmetic(const char *input) {
    int i = 0;
    char token_buf[128];

    while (input[i] != '\0') {
        if (isspace((unsigned char)input[i])) {
            i++;
            continue;
        }

        if (isdigit((unsigned char)input[i])) {
            int num_type = 0;
            if (classify_number(input, &i, token_buf, &num_type)) {
                if (num_type == 1) {
                    printf("SCIENTIFIC: %s\n", token_buf);
                } else if (num_type == 2) {
                    printf("FLOAT: %s\n", token_buf);
                } else if (num_type == 3) {
                    printf("INTEGER: %s\n", token_buf);
                }
            } else {
                printf("INVALID NUMERIC CONSTANT: %s\n", token_buf);
            }
        } else if (input[i] == '+') {
            printf("PLUS: +\n");
            i++;
        } else if (input[i] == '-') {
            printf("MINUS: -\n");
            i++;
        } else if (input[i] == '*') {
            printf("MULTIPLY: *\n");
            i++;
        } else if (input[i] == '/') {
            printf("DIVIDE: /\n");
            i++;
        } else if (input[i] == '(') {
            printf("LPAREN: (\n");
            i++;
        } else if (input[i] == ')') {
            printf("RPAREN: )\n");
            i++;
        } else {
            printf("INVALID: %c\n", input[i]);
            i++;
        }
    }
}

int main(void) {
    char input[1024];
    printf("Enter expression:\n");
    if (fgets(input, sizeof(input), stdin)) {
        size_t len = strlen(input);
        while (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {

node *addleaf ( char c )
{
    node *p;
    p = (node *)malloc(sizeof(node));
    p -> symb = c;
    if ((c >= '0') && (c <= '9')) p -> val = c - '0';
    p -> nc = 0;
    p -> C = NULL;
    return p;
}

node *addnodeN ( node *C1, node *C2 )
{
    node *p;
    p = (node *)malloc(sizeof(node));
    p -> symb = 'N';
    if (C2 != NULL) {
        p -> nc = 2;
        p -> C = (node **)malloc(2 * sizeof(node *));
        (p -> C)[0] = C1;
        (p -> C)[1] = C2;
    } else {
        p -> nc = 1;
        p -> C = (node **)malloc(sizeof(node *));
        (p -> C)[0] = C1;
    }
    return p;
}

node *addnodeM ( node *C1, node *C2 )

void setatt ( node *p )
{
    switch (p -> symb) {
    case 'S':
        if (p -> nc == 1) {
            (p -> C)[0] -> inh = PLUS;
            setatt((p->C)[0]);
        } else {
            (p -> C)[1] -> inh = ((p -> C)[0] -> symb == '+') ? PLUS : MINUS;
            setatt((p->C)[1]);
        }
        break;
    case 'P':
        (p -> C)[0] -> inh = p -> inh;
        setatt((p->C)[0]);
        if (p -> nc == 3) {
            (p -> C)[2] -> inh = ((p -> C)[1] -> symb == '+') ? PLUS : MINUS;
            setatt((p->C)[2]);
        }
        break;
    case 'T':
        if (p -> nc == 1) {
            if ((p -> C)[0] -> symb == 'N') (p -> C)[0] -> inh = 0;
            setatt((p->C)[0]);
        } else {
            setatt((p->C)[0]);
            setatt((p->C)[1]);
        }
        break;
    case 'X':
        if (p -> nc == 3) {
            setatt((p->C)[2]);
        }
        break;
    case 'N':
        if (p -> nc == 1) {
            p -> val = (p -> C)[0] -> val;
        } else {
            (p -> C)[1] -> inh = (p -> C)[0] -> val;
            setatt((p->C)[1]);
            p -> val = (p -> C)[1] -> val;
        }
        break;
    case 'M':
        if (p -> nc == 1) {
            p -> val = 10 * (p -> inh) + (p -> C)[0] -> val;
        } else {
            (p -> C)[1] -> inh = 10 * (p -> inh) + (p -> C)[0] -> val;
            setatt((p->C)[1]);
            p -> val = (p -> C)[1] -> val;
        }
        break;
    }
}

void printPT ( node *p, int l )
{
    int i;
    for (i=0; i<l-1; ++i) printf("   ");
    if (l > 1) printf("==> "); else printf(" ");
    printf("%c [", p -> symb);
    if ((p -> symb == 'P') || (p -> symb == 'T'))
        printf("inh = %c", (p -> inh == PLUS) ? '+' : '-');
    else if ((p -> symb >= '0') && (p -> symb <= '9'))
        printf("val = %d", p -> val);
    else if (p -> symb == 'N')
        printf("val = %d", p -> val);
    else if (p -> symb == 'M')
        printf("inh = %d, val = %d", p -> inh, p -> val);
    printf("]\n");
    for (i=0; i<p->nc; ++i) printPT((p->C)[i],l+1);
}

int pwr ( int a, int e )
{
    int t;
    if (e == 0) return 1;
    t = pwr(a, e/2);
    t = t * t;
    if (e % 2 == 1) t *= a;
    return t;
}

int evalpoly ( node *p, int xval )
{
    if (p -> symb == 'S') {
        if (p -> nc == 1) return evalpoly((p -> C)[0], xval);
        else return evalpoly((p -> C)[1], xval);
    }
    if (p -> symb == 'P') {
        if (p -> nc == 1) return evalpoly((p -> C)[0], xval);
        else return evalpoly((p -> C)[0], xval) + evalpoly((p -> C)[2], xval);
    }
    if (p -> symb == 'T') {
        if (p -> inh == PLUS) {
            if (p -> nc == 1) return evalpoly((p -> C)[0], xval);
            else return evalpoly((p -> C)[0], xval) * evalpoly((p -> C)[1], xval);
        } else {
            if (p -> nc == 1) return -(evalpoly((p -> C)[0], xval));
            else return -(evalpoly((p -> C)[0], xval)) * evalpoly((p -> C)[1], xval);
        }
    }
    if (p -> symb == 'X') {
        if (p -> nc == 1) return xval;
        else return pwr(xval, evalpoly((p -> C)[2], xval));
    }
    if ((p -> symb == 'N') || (p -> symb == 'M')) return p -> val;
    if (p -> symb >= '1') return 1;
    return 0;
}

        printf("%d", C);
    } else if (E == 1) {
        if (C != 1) printf("%d", C);
        printf("x");
    } else {
        if (C != 1) printf("%d", C);
        printf("x^%d", E);
    }

        if (p -> nc == 1) printderivative((p -> C)[0]);
        else printderivative((p -> C)[1]);
    } else if (p -> symb == 'P') {
        if (p -> nc == 1) printderivative((p -> C)[0]);
        else {
            printderivative((p -> C)[0]);
            printderivative((p -> C)[2]);
        }
    } else if (p -> symb == 'T') {
        if (p -> nc == 1) {
            if ((p -> C)[0] -> symb == 'X') {
                if (p -> inh == PLUS) printf(" + "); else printf(" - ");
                if (((p -> C)[0]) -> nc == 1) prnterm(1,0);
                else prnterm(((p -> C)[0] -> C)[2] -> val, ((p -> C)[0] -> C)[2] -> val - 1);

            else prnterm(((p -> C)[0] -> val) * ((q -> C)[2] -> val), (q -> C)[2] -> val - 1);
        }
    }
}

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

typedef enum {
    TOK_ID,
    TOK_NUM,
    TOK_PLUS,
    TOK_MINUS,
    TOK_MUL,
    TOK_DIV,
    TOK_LPAREN,
    TOK_RPAREN,
    TOK_END,
    TOK_INVALID
} TokenType;

typedef struct {
    TokenType type;
    char text[64];
} Token;

static Token tokens[1024];
static int token_count = 0;
static int cur_token = 0;

static char postfix_output[4096];
static int postfix_pos = 0;

static void emit(const char *str) {
    int len = (int)strlen(str);
    if (postfix_pos + len < (int)sizeof(postfix_output) - 1) {
        strcpy(postfix_output + postfix_pos, str);
        postfix_pos += len;
    }
}

static int tokenize(const char *input) {
    token_count = 0;
    cur_token = 0;
    postfix_pos = 0;
    postfix_output[0] = '\0';
    int i = 0;

    while (input[i] != '\0') {
        if (isspace((unsigned char)input[i])) {
            i++;
            continue;
        }

        if (isalpha((unsigned char)input[i]) || input[i] == '_') {
            int start = i;
            while (isalnum((unsigned char)input[i]) || input[i] == '_') i++;
            int len = i - start;
            tokens[token_count].type = TOK_ID;
            strncpy(tokens[token_count].text, input + start, len);
            tokens[token_count].text[len] = '\0';
            token_count++;
        } else if (isdigit((unsigned char)input[i])) {
            int start = i;
            while (isdigit((unsigned char)input[i])) i++;
            int len = i - start;
            tokens[token_count].type = TOK_NUM;
            strncpy(tokens[token_count].text, input + start, len);
            tokens[token_count].text[len] = '\0';
            token_count++;
        } else if (input[i] == '+') {
            tokens[token_count].type = TOK_PLUS;
            strcpy(tokens[token_count].text, "+");
            token_count++;
            i++;
        } else if (input[i] == '-') {
            tokens[token_count].type = TOK_MINUS;
            strcpy(tokens[token_count].text, "-");
            token_count++;
            i++;
        } else if (input[i] == '*') {
            tokens[token_count].type = TOK_MUL;
            strcpy(tokens[token_count].text, "*");
            token_count++;
            i++;
        } else if (input[i] == '/') {
            tokens[token_count].type = TOK_DIV;
            strcpy(tokens[token_count].text, "/");
            token_count++;
            i++;
        } else if (input[i] == '(') {
            tokens[token_count].type = TOK_LPAREN;
            strcpy(tokens[token_count].text, "(");
            token_count++;
            i++;
        } else if (input[i] == ')') {
            tokens[token_count].type = TOK_RPAREN;
            strcpy(tokens[token_count].text, ")");
            token_count++;
            i++;
        } else {
            tokens[token_count].type = TOK_INVALID;
            token_count++;
            return 0;
        }
    }

    tokens[token_count].type = TOK_END;
    tokens[token_count].text[0] = '\0';
    token_count++;
    return 1;
}

/*
Syntax-Directed Translation Grammar with Semantic Actions:
  E -> T E'
  E' -> + T { emit("+") } E'
      | - T { emit("-") } E'
      | ε
  T -> F T'
  T' -> * F { emit("*") } T'
      | / F { emit("/") } T'
      | ε
  F -> ( E )
     | ID  { emit(id) }
     | NUM { emit(num) }
*/

static int parse_E(void);
static int parse_T(void);
static int parse_F(void);

static int parse_F(void) {
    if (tokens[cur_token].type == TOK_ID || tokens[cur_token].type == TOK_NUM) {
        emit(tokens[cur_token].text); // Semantic Action: emit identifier or number
        cur_token++;
        return 1;
    }
    if (tokens[cur_token].type == TOK_LPAREN) {
        cur_token++;
        if (!parse_E()) return 0;
        if (tokens[cur_token].type != TOK_RPAREN) return 0;
        cur_token++;
        return 1;
    }
    return 0;
}

static int parse_T(void) {
    if (!parse_F()) return 0;
    while (tokens[cur_token].type == TOK_MUL || tokens[cur_token].type == TOK_DIV) {
        TokenType op = tokens[cur_token].type;
        cur_token++;
        if (!parse_F()) return 0;
        // Semantic Action executed after operand (postfix order):
        if (op == TOK_MUL) emit("*");
        else emit("/");
    }
    return 1;
}

static int parse_E(void) {
    if (!parse_T()) return 0;
    while (tokens[cur_token].type == TOK_PLUS || tokens[cur_token].type == TOK_MINUS) {
        TokenType op = tokens[cur_token].type;
        cur_token++;
        if (!parse_T()) return 0;
        // Semantic Action executed after operand (postfix order):
        if (op == TOK_PLUS) emit("+");
        else emit("-");
    }
    return 1;
}

int translate_infix_to_postfix(const char *input, char *result) {
    if (!tokenize(input)) return 0;
    if (tokens[0].type == TOK_END) return 0;
    if (!parse_E()) return 0;
    if (tokens[cur_token].type != TOK_END) return 0;
    strcpy(result, postfix_output);
    return 1;
}

int main(void) {
    char input[256];
    char result[4096];

    printf("Enter infix expression: ");
    if (fgets(input, sizeof(input), stdin)) {
        size_t len = strlen(input);
        while (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {
            input[--len] = '\0';
        }

        if (translate_infix_to_postfix(input, result)) {
            printf("Postfix expression: %s\n", result);
        } else {
            fprintf(stderr, "Error: Invalid arithmetic expression\n");
        }
    }
    return 0;
}

    if (l > 1) printf("==> "); else printf(" ");
    printf("%c [", p->symb);
    if ((p->symb == 'P') || (p->symb == 'T'))
        printf("inh = %c", (p->inh == PLUS) ? '+' : '-');
    else if ((p->symb >= '0') && (p->symb <= '9'))
        printf("val = %d", p->val);
    else if (p->symb == 'N')
        printf("val = %d", p->val);
    else if (p->symb == 'M')
        printf("inh = %d, val = %d", p->inh, p->val);
    printf("]\n");
    for (i = 0; i < p->nc; ++i) printPT((p->C)[i], l + 1);
}

int pwr(int a, int e) {
    int t;
    if (e == 0) return 1;
    t = pwr(a, e / 2);
    t = t * t;
    if (e % 2 == 1) t *= a;
    return t;
}

int evalpoly(node *p, int xval) {
    if (!p) return 0;
    if (p->symb == 'S') {
        if (p->nc == 1) return evalpoly((p->C)[0], xval);
        else return evalpoly((p->C)[1], xval);
    }
    if (p->symb == 'P') {
        if (p->nc == 1) return evalpoly((p->C)[0], xval);
        else return evalpoly((p->C)[0], xval) + evalpoly((p->C)[2], xval);
    }
    if (p->symb == 'T') {
        if (p->inh == PLUS) {
            if (p->nc == 1) return evalpoly((p->C)[0], xval);
            else return evalpoly((p->C)[0], xval) * evalpoly((p->C)[1], xval);
        } else {
            if (p->nc == 1) return -(evalpoly((p->C)[0], xval));
            else return -(evalpoly((p->C)[0], xval)) * evalpoly((p->C)[1], xval);
        }
    }
    if (p->symb == 'X') {
        if (p->nc == 1) return xval;
        else return pwr(xval, evalpoly((p->C)[2], xval));
    }
    if ((p->symb == 'N') || (p->symb == 'M')) return p->val;
    if (p->symb >= '1') return 1;
    return 0;
}

void prnterm(int C, int E) {
    if (C == 0) return;
    if (E == 0) {
        printf("%d", C);
    } else if (E == 1) {
        if (C != 1) printf("%d", C);
        printf("x");
    } else {
        if (C != 1) printf("%d", C);
        printf("x^%d", E);
    }
}

void printderivative(node *p) {
    node *q;
    if (!p) return;
    if (p->symb == 'S') {
        if (p->nc == 1) printderivative((p->C)[0]);
        else printderivative((p->C)[1]);
    } else if (p->symb == 'P') {
        if (p->nc == 1) printderivative((p->C)[0]);
        else {
            printderivative((p->C)[0]);
            printderivative((p->C)[2]);
        }
    } else if (p->symb == 'T') {
        if (p->nc == 1) {
            if ((p->C)[0]->symb == 'X') {
                if (p->inh == PLUS) printf(" + "); else printf(" - ");
                if (((p->C)[0])->nc == 1) prnterm(1, 0);
                else prnterm(((p->C)[0]->C)[2]->val, ((p->C)[0]->C)[2]->val - 1);
            }
        } else {
            if (p->inh == PLUS) printf(" + "); else printf(" - ");
            q = (p->C)[1];
            if (q->nc == 1) prnterm((p->C)[0]->val, 0);
            else prnterm(((p->C)[0]->val) * ((q->C)[2]->val), (q->C)[2]->val - 1);
        }
    }
}

/* Lexer and Parser for polynomial expression string */
static const char *cursor;

static void skip_ws() {
    while (*cursor == ' ' || *cursor == '\t' || *cursor == '\r' || *cursor == '\n') cursor++;
}

static node *parse_N_num(int val) {
    char s[32];
    sprintf(s, "%d", val);
    int len = (int)strlen(s);
    if (len == 1) {
        return addnodeN(addleaf(s[0]), NULL);
    }
    // Build M chain
    node *m = addleaf(s[len - 1]);
    for (int i = len - 2; i >= 1; i--) {
        m = addnodeM(addleaf(s[i]), m);
    }
    return addnodeN(addleaf(s[0]), m);
}

static node *parse_polynomial_string(const char *str) {
    cursor = str;
    skip_ws();
    char lead_sign = '\0';
    if (*cursor == '+' || *cursor == '-') {
        lead_sign = *cursor;
        cursor++;
        skip_ws();
    }

    node *head_p = NULL;

    while (*cursor != '\0') {
        skip_ws();
        if (*cursor == '\0') break;

        char op_sign = '+';
        if (head_p != NULL) {
            if (*cursor == '+' || *cursor == '-') {
                op_sign = *cursor;
                cursor++;
                skip_ws();
            }
        }

        // Parse Term T: could be [N] [x [^ N]]
        int has_coef = 0;
        int coef = 1;
        if (isdigit((unsigned char)*cursor)) {
            has_coef = 1;
            coef = 0;
            while (isdigit((unsigned char)*cursor)) {
                coef = coef * 10 + (*cursor - '0');
                cursor++;
            }
            skip_ws();
        }

        node *node_t = NULL;
        if (*cursor == 'x') {
            cursor++;
            skip_ws();
            node *node_x = NULL;
            if (*cursor == '^') {
                cursor++;
                skip_ws();
                int exp_val = 0;
                while (isdigit((unsigned char)*cursor)) {
                    exp_val = exp_val * 10 + (*cursor - '0');
                    cursor++;
                }
                node_x = addnodeX(addleaf('x'), parse_N_num(exp_val));
            } else {
                node_x = addnodeX(addleaf('x'), NULL);
            }

            if (has_coef) {
                node_t = addnodeT('B', parse_N_num(coef), node_x);
            } else {
                node_t = addnodeT('X', node_x, NULL);
            }
        } else if (has_coef) {
            node_t = addnodeT('N', parse_N_num(coef), NULL);
        }

        if (!node_t) break;

        if (head_p == NULL) {
            head_p = addnodeP('\0', node_t, NULL);
        } else {
            head_p = addnodeP(op_sign, head_p, node_t);
        }
        skip_ws();
    }

    return addnodeS(lead_sign, head_p);
}

int main(int argc, char *argv[]) {
    char input[1024];
    const char *default_poly = "-506x^7 + 9x^10 + x - 243015 - x^3 + 876 x^2 + 1";

    printf("Enter polynomial (or press enter for default: %s):\n", default_poly);
    if (fgets(input, sizeof(input), stdin)) {
        size_t len = strlen(input);
        while (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {
            input[--len] = '\0';
        }
    } else {
        input[0] = '\0';
    }

    const char *poly_str = (strlen(input) > 0) ? input : default_poly;
    printf("\nParsing polynomial: %s\n", poly_str);

    PT = parse_polynomial_string(poly_str);
    setatt(PT);

    printf("+++ The annotated parse tree is\n");
    printPT(PT, 1);
    printf("\n");

    for (int xval = -5; xval <= 5; ++xval) {
        printf("+++ f(%2d) = %10d\n", xval, evalpoly(PT, xval));
    }

    printf("\n+++ f'(x) = ");
    printderivative(PT);
    printf("\n");

    return 0;
}

/*
================================================================================
EXPERIMENT 10: EXPRESSION EVALUATION USING AST & SYMBOL TABLE
================================================================================
QUESTIONS:
Implement an expression evaluation program using Lex and Yacc to evaluate 
arithmetic expressions containing identifiers, integer constants, and operators 
such as addition (+), subtraction (-), multiplication (*), division (/), 
modulus (%), and exponentiation (^).

The program should maintain a symbol table to store identifiers and their values, 
construct an expression tree using leaf nodes and operator nodes, and evaluate the 
expression tree to obtain the final result. It should handle operator precedence, 
parentheses, and variable assignments, while displaying appropriate error 
messages for invalid identifiers, uninitialized variables, and invalid expressions.

Sample Session (from Lab Record Ex 10 Output):
  Input:  (set a 5)
  Output: Variable a is set to 5

  Input:  (set b 9)
  Output: Variable b is set to 9

  Input:  (set c (+ a b))
  Output: Variable c is set to 14

  Input:  (* a b)
  Output: Standalone expression evaluates to 45

  Input:  (set a
  Output: Error: syntax error

--------------------------------------------------------------------------------
COMPANION LEX & YACC FILES INCLUDED IN EXPERIMENT RECORD:
  - runprog.c
  - expr.y
  - expr.l
================================================================================
*/

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define NUM 1
#define ID  2
#define OP  3

#define ADD 101
#define SUB 102
#define MUL 103
#define DIV 104
#define REM 105
#define POW 106

/* Symbol Table Definition */
typedef struct {
    int type;        /* NUM or ID */
    char *data;      /* Name of identifier or string of number */
    int value;       /* Numeric value */
    int init;        /* Initialized (1) or not (0) */
} stentry;

static stentry ST[1024];
static int nsymb = 0;

/* Expression Tree Definition */
typedef struct _node {
    int type;        /* OP, ID, or NUM */
    int data;        /* Opcode or index in symbol table */
    struct _node *L, *R;
} node;

int addsymbol(int type, const char *text) {
    int i;
    for (i = 0; i < nsymb; ++i) {
        if (!strcmp(ST[i].data, text)) return i;
    }
    i = nsymb;
    ++nsymb;
    ST[i].type = type;
    ST[i].data = (char *)malloc((1 + strlen(text)) * sizeof(char));
    strcpy(ST[i].data, text);
    if (type == ID) {
        ST[i].value = 0;
        ST[i].init = 0;
    } else if (type == NUM) {
        ST[i].value = atoi(text);
        ST[i].init = 1;
    } else {
        fprintf(stderr, "Error: Unknown type %d for symbol table\n", type);
        --nsymb;
        return -1;
    }
    return i;
}

int setsymbol(int i, int val) {
    if ((i < 0) || (i >= nsymb)) {
        fprintf(stderr, "Error: Invalid index %d in symbol table\n", i);
        return -1;
    }
    if (ST[i].type != ID) {
        fprintf(stderr, "Error: Attempt to set non-ID in symbol table\n");
        return -1;
    }
    ST[i].value = val;
    ST[i].init = 1;
    return i;
}

int getsymbol(const char *id, int *err) {
    int i;
    for (i = 0; i < nsymb; ++i) {
        if (!strcmp(ST[i].data, id)) {
            if (ST[i].init == 0) {
                fprintf(stderr, "Error: Attempt to read uninitialized value\n");
                if (err) *err = 1;
                return 0;
            }
            return ST[i].value;
        }
    }
    fprintf(stderr, "Error: Invalid ID %s\n", id);
    if (err) *err = 1;
    return 0;
}

int getvalue(int i, int *err) {
    if ((i < 0) || (i >= nsymb)) {
        fprintf(stderr, "Error: Unknown index %d in symbol table\n", i);
        if (err) *err = 1;
        return 0;
    }
    if (ST[i].init == 0) {
        fprintf(stderr, "Error: Attempt to read uninitialized value\n");
        if (err) *err = 1;
        return 0;
    }
    return ST[i].value;
}

node *addleaf(int i) {
    if ((i < 0) || (i >= nsymb)) {
        fprintf(stderr, "Invalid index %d in symbol table\n", i);
        return NULL;
    }
    node *p = (node *)malloc(sizeof(node));
    p->type = ST[i].type;
    p->data = i;
    p->L = p->R = NULL;
    return p;
}

node *addnode(int op, node *left, node *right) {
    node *p = (node *)malloc(sizeof(node));
    p->type = OP;
    p->data = op;
    p->L = left;
    p->R = right;
    return p;
}

int intpow(int a, int e) {
    int prod = 1, a2pow = a;
    if (e < 0) return 0;
    while (e > 0) {
        if (e % 2) prod *= a2pow;
        e /= 2;
        a2pow *= a2pow;
    }
    return prod;
}

int evaltree(node *p, int *err) {
    int val = 0, lval = 0, rval = 0;
    if (p == NULL) {
        fprintf(stderr, "Error: Empty expression tree\n");
        if (err) *err = 1;
        return 0;
    }
    if ((p->type == ID) || (p->type == NUM)) {
        val = getvalue(p->data, err);
    } else if (p->type == OP) {
        lval = evaltree(p->L, err);
        rval = evaltree(p->R, err);
        switch (p->data) {
            case ADD: val = lval + rval; break;
            case SUB: val = lval - rval; break;
            case MUL: val = lval * rval; break;
            case DIV:
                if (rval == 0) {
                    fprintf(stderr, "Error: Division by zero\n");
                    if (err) *err = 1;
                    val = 0;
                } else {
                    val = lval / rval;
                }
                break;
            case REM:
                if (rval == 0) {
                    fprintf(stderr, "Error: Modulo by zero\n");
                    if (err) *err = 1;
                    val = 0;
                } else {
                    val = lval % rval;
                }
                break;
            case POW: val = intpow(lval, rval); break;
            default:
                fprintf(stderr, "Error: Unknown opcode %d\n", p->data);
                if (err) *err = 1;
                val = 0;
                break;
        }
    } else {
        fprintf(stderr, "Error: Unknown node type in expression tree\n");
        if (err) *err = 1;
        val = 0;
    }
    free(p);
    return val;
}

/* Lexer and Parser for (set ID val) and (op arg arg) */
typedef enum {
    TOK_LP,
    TOK_RP,
    TOK_SET,
    TOK_OP_ADD,
    TOK_OP_SUB,
    TOK_OP_MUL,
    TOK_OP_DIV,
    TOK_OP_REM,
    TOK_OP_POW,
    TOK_ID_STR,
    TOK_NUM_STR,
    TOK_EOF_OR_ERR
} ParseToken;

static const char *parse_ptr;

static void skip_spaces() {
    while (*parse_ptr == ' ' || *parse_ptr == '\t' || *parse_ptr == '\r' || *parse_ptr == '\n') parse_ptr++;
}

static ParseToken next_token(char *out_val) {
    skip_spaces();
    if (*parse_ptr == '\0') return TOK_EOF_OR_ERR;

    if (*parse_ptr == '(') {
        parse_ptr++;
        return TOK_LP;
    }
    if (*parse_ptr == ')') {
        parse_ptr++;
        return TOK_RP;
    }

    if (*parse_ptr == '*' && *(parse_ptr + 1) == '*') {
        parse_ptr += 2;
        return TOK_OP_POW;
    }
    if (*parse_ptr == '^') {
        parse_ptr++;
        return TOK_OP_POW;
    }
    if (*parse_ptr == '+') {
        parse_ptr++;
        return TOK_OP_ADD;
    }
    if (*parse_ptr == '-' && !isdigit((unsigned char)*(parse_ptr + 1))) {
        parse_ptr++;
        return TOK_OP_SUB;
    }
    if (*parse_ptr == '*') {
        parse_ptr++;
        return TOK_OP_MUL;
    }
    if (*parse_ptr == '/') {
        parse_ptr++;
        return TOK_OP_DIV;
    }
    if (*parse_ptr == '%') {
        parse_ptr++;
        return TOK_OP_REM;
    }

    if (isdigit((unsigned char)*parse_ptr) || (*parse_ptr == '-' && isdigit((unsigned char)*(parse_ptr + 1)))) {
        int i = 0;
        if (*parse_ptr == '-') out_val[i++] = *parse_ptr++;
        while (isdigit((unsigned char)*parse_ptr)) {
            out_val[i++] = *parse_ptr++;
        }
        out_val[i] = '\0';
        return TOK_NUM_STR;
    }

    if (isalpha((unsigned char)*parse_ptr) || *parse_ptr == '_') {
        int i = 0;
        while (isalnum((unsigned char)*parse_ptr) || *parse_ptr == '_') {
            out_val[i++] = *parse_ptr++;
        }
        out_val[i] = '\0';
        if (strcmp(out_val, "set") == 0) return TOK_SET;
        return TOK_ID_STR;
    }

    return TOK_EOF_OR_ERR;
}

static node *parse_expr_internal(int *err);

static node *parse_arg_internal(int *err) {
    const char *save = parse_ptr;
    char token_text[128];
    ParseToken tok = next_token(token_text);
    if (tok == TOK_NUM_STR) {
        return addleaf(addsymbol(NUM, token_text));
    }
    if (tok == TOK_ID_STR) {
        return addleaf(addsymbol(ID, token_text));
    }
    if (tok == TOK_LP) {
        parse_ptr = save; // rewind to '('
        return parse_expr_internal(err);
    }
    *err = 1;
    return NULL;
}

static node *parse_expr_internal(int *err) {
    char token_text[128];
    ParseToken tok = next_token(token_text);
    if (tok != TOK_LP) {
        *err = 1;
        return NULL;
    }

    tok = next_token(token_text);
    int opcode = 0;
    switch (tok) {
        case TOK_OP_ADD: opcode = ADD; break;
        case TOK_OP_SUB: opcode = SUB; break;
        case TOK_OP_MUL: opcode = MUL; break;
        case TOK_OP_DIV: opcode = DIV; break;
        case TOK_OP_REM: opcode = REM; break;
        case TOK_OP_POW: opcode = POW; break;
        default:
            *err = 1;
            return NULL;
    }

    node *left = parse_arg_internal(err);
    if (*err || !left) return NULL;
    node *right = parse_arg_internal(err);
    if (*err || !right) return NULL;

    tok = next_token(token_text);
    if (tok != TOK_RP) {
        *err = 1;
        return NULL;
    }

    return addnode(opcode, left, right);
}

void process_line(const char *line) {
    parse_ptr = line;
    skip_spaces();
    if (*parse_ptr == '\0') return;

    char token_text[128];
    const char *save_start = parse_ptr;
    ParseToken tok = next_token(token_text);

    if (tok != TOK_LP) {
        fprintf(stderr, "Error: syntax error\n");
        return;
    }

    const char *after_lp = parse_ptr;
    tok = next_token(token_text);

    if (tok == TOK_SET) {
        // (set ID NUM|ID|expr )
        tok = next_token(token_text);
        if (tok != TOK_ID_STR) {
            fprintf(stderr, "Error: syntax error\n");
            return;
        }
        char var_name[64];
        strcpy(var_name, token_text);
        int sym_idx = addsymbol(ID, var_name);

        const char *arg_save = parse_ptr;
        tok = next_token(token_text);
        int eval_err = 0;
        int value = 0;

        if (tok == TOK_NUM_STR) {
            value = atoi(token_text);
            tok = next_token(token_text);
            if (tok != TOK_RP) {
                fprintf(stderr, "Error: syntax error\n");
                return;
            }
        } else if (tok == TOK_ID_STR) {
            value = getsymbol(token_text, &eval_err);
            tok = next_token(token_text);
            if (tok != TOK_RP || eval_err) {
                if (!eval_err) fprintf(stderr, "Error: syntax error\n");
                return;
            }
        } else if (tok == TOK_LP) {
            parse_ptr = arg_save;
            node *tree = parse_expr_internal(&eval_err);
            if (eval_err || !tree) {
                fprintf(stderr, "Error: syntax error\n");
                return;
            }
            tok = next_token(token_text);
            if (tok != TOK_RP) {
                fprintf(stderr, "Error: syntax error\n");
                return;
            }
            value = evaltree(tree, &eval_err);
            if (eval_err) return;
        } else {
            fprintf(stderr, "Error: syntax error\n");
            return;
        }

        setsymbol(sym_idx, value);
        printf("Variable %s is set to %d\n", ST[sym_idx].data, ST[sym_idx].value);
    } else {
        // Standalone expression: (op arg arg)
        parse_ptr = save_start;
        int eval_err = 0;
        node *tree = parse_expr_internal(&eval_err);
        if (eval_err || !tree) {
            fprintf(stderr, "Error: syntax error\n");
            return;
        }
        int val = evaltree(tree, &eval_err);
        if (!eval_err) {
            printf("Standalone expression evaluates to %d\n", val);
        }
    }
}

/*
================================================================================
EXPERIMENT 10: PARSER IMPLEMENTATION (PREDICTIVE LL(1) PARSER)
================================================================================
AIM:
To study and implement a predictive parser (LL(1) parser) using a given grammar, 
construct the predictive parsing table using FIRST and FOLLOW sets, and verify 
the acceptance of valid input strings while identifying invalid strings.

INTRODUCTION:
A predictive parser is a top-down parser used in compiler design to determine 
whether an input string belongs to a given grammar. It uses a parsing table and a 
stack to select the appropriate production rules based on the current input symbol 
and the top of the stack. An LL(1) parser scans the input from left to right, 
produces a leftmost derivation, and uses one lookahead symbol to make parsing 
decisions.

Standard Arithmetic Grammar (Eliminating Left Recursion):
  1. E  -> T E'
  2. E' -> + T E'
  3. E' -> e (epsilon)
  4. T  -> F T'
  5. T' -> * F T'
  6. T' -> e (epsilon)
  7. F  -> ( E )
  8. F  -> id

FIRST Sets:
  FIRST(F)  = { (, id }
  FIRST(T') = { *, e }
  FIRST(T)  = { (, id }
  FIRST(E') = { +, e }
  FIRST(E)  = { (, id }

FOLLOW Sets:
  FOLLOW(E)  = { $, ) }
  FOLLOW(E') = { $, ) }
  FOLLOW(T)  = { +, $, ) }
  FOLLOW(T') = { +, $, ) }
  FOLLOW(F)  = { *, +, $, ) }

LL(1) Parsing Table M[A, a]:
-------------------------------------------------------------------------
Non-Term |      id      |      +       |      *       |      (       |      )       |      $
-------------------------------------------------------------------------
   E     |   E->TE'     |              |              |   E->TE'     |              |
   E'    |              |   E'->+TE'   |              |              |    E'->e     |    E'->e
   T     |   T->FT'     |              |              |   T->FT'     |              |
   T'    |              |    T'->e     |   T'->*FT'   |              |    T'->e     |    T'->e
   F     |    F->id     |              |              |   F->(E)     |              |
-------------------------------------------------------------------------
================================================================================
*/

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_STACK 100

/* Grammar Non-Terminals: E=0, E'=1, T=2, T'=3, F=4 */
/* Terminals: id=0, +=1, *=2, (=3, )=4, $=5 */

const char *non_terminals[] = {"E", "E'", "T", "T'", "F"};
const char *terminals[] = {"id", "+", "*", "(", ")", "$"};

/*
Production representations:
  0: (none)
  1: E -> T E'
  2: E' -> + T E'
  3: E' -> e
  4: T -> F T'
  5: T' -> * F T'
  6: T' -> e
  7: F -> ( E )
  8: F -> id
*/
const char *production_str[] = {
    "Error",
    "E -> T E'",
    "E' -> + T E'",
    "E' -> e",
    "T -> F T'",
    "T' -> * F T'",
    "T' -> e",
    "F -> ( E )",
    "F -> id"
};

int parsing_table[5][6] = {
    /*          id    +    *    (    )    $   */
    /* E  */  {  1,   0,   0,   1,   0,   0 },
    /* E' */  {  0,   2,   0,   0,   3,   3 },
    /* T  */  {  4,   0,   0,   4,   0,   0 },
    /* T' */  {  0,   6,   5,   0,   6,   6 },
    /* F  */  {  8,   0,   0,   7,   0,   0 }
};

int get_terminal_index(const char *tok) {
    if (strcmp(tok, "id") == 0) return 0;
    if (strcmp(tok, "+") == 0) return 1;
    if (strcmp(tok, "*") == 0) return 2;
    if (strcmp(tok, "(") == 0) return 3;
    if (strcmp(tok, ")") == 0) return 4;
    if (strcmp(tok, "$") == 0) return 5;
    return -1;
}

/* Parse table-driven LL(1) simulator */
int parse_string(const char *input_tokens[], int num_tokens) {
    char stack[MAX_STACK][8];
    int top = 0;

    // Push initial markers: $ and start symbol E
    strcpy(stack[top++], "$");
    strcpy(stack[top++], "E");

    int ip = 0;

    printf("\n%-25s %-25s %-25s\n", "Stack", "Input", "Action");
    printf("-------------------------------------------------------------------------\n");

    while (top > 0) {
        // Print stack
        char stack_str[128] = "";
        for (int i = 0; i < top; i++) {
            strcat(stack_str, stack[i]);
            strcat(stack_str, " ");
        }

        // Print remaining input
        char input_str[128] = "";
        for (int i = ip; i < num_tokens; i++) {
            strcat(input_str, input_tokens[i]);
            strcat(input_str, " ");
        }

        char *X = stack[top - 1];
        const char *a = input_tokens[ip];
        int a_idx = get_terminal_index(a);

        if (a_idx == -1) {
            printf("%-25s %-25s Error: Invalid token '%s'\n", stack_str, input_str, a);
            return 0;
        }

        // Check if X is terminal or '$'
        int is_terminal = 0;
        for (int t = 0; t < 6; t++) {
            if (strcmp(X, terminals[t]) == 0) {
                is_terminal = 1;
                break;
            }
        }

        if (is_terminal) {
            if (strcmp(X, a) == 0) {
                char action[64];
                sprintf(action, "Match %s", a);
                printf("%-25s %-25s %-25s\n", stack_str, input_str, action);
                top--; // pop X
                ip++;  // advance input
                if (strcmp(X, "$") == 0) {
                    return 1; // accepted
                }
            } else {
                printf("%-25s %-25s Error: Mismatch, expected '%s' got '%s'\n", stack_str, input_str, X, a);
                return 0;
            }
        } else {
            // X is non-terminal
            int nt_idx = -1;
            for (int nt = 0; nt < 5; nt++) {
                if (strcmp(X, non_terminals[nt]) == 0) {
                    nt_idx = nt;
                    break;
                }
            }

            if (nt_idx == -1) {
                printf("%-25s %-25s Error: Unknown symbol '%s'\n", stack_str, input_str, X);
                return 0;
            }

            int rule = parsing_table[nt_idx][a_idx];
            if (rule == 0) {
                printf("%-25s %-25s Error: No rule in M[%s, %s]\n", stack_str, input_str, X, a);
                return 0;
            }

            printf("%-25s %-25s Output: %s\n", stack_str, input_str, production_str[rule]);
            top--; // pop non-terminal

            // Push right-hand side in reverse order
            switch (rule) {
                case 1: // E -> T E'
                    strcpy(stack[top++], "E'");
                    strcpy(stack[top++], "T");
                    break;
                case 2: // E' -> + T E'
                    strcpy(stack[top++], "E'");
                    strcpy(stack[top++], "T");
                    strcpy(stack[top++], "+");
                    break;
                case 3: // E' -> e (nothing pushed)
                    break;
                case 4: // T -> F T'
                    strcpy(stack[top++], "T'");
                    strcpy(stack[top++], "F");
                    break;
                case 5: // T' -> * F T'
                    strcpy(stack[top++], "T'");
                    strcpy(stack[top++], "F");
                    strcpy(stack[top++], "*");
                    break;
                case 6: // T' -> e (nothing pushed)
                    break;
                case 7: // F -> ( E )
                    strcpy(stack[top++], ")");
                    strcpy(stack[top++], "E");
                    strcpy(stack[top++], "(");
                    break;
                case 8: // F -> id
                    strcpy(stack[top++], "id");
                    break;
            }
        }
    }

    return 0;
}

int addsymbol ( int type, char *text )
{
    int i;
    for (i=0; i<nsymb; ++i) if (!strcmp(ST[i].data,text)) return i;
    i = nsymb;
    ++nsymb;
    ST[i].type = type;
    ST[i].data = (char *)malloc((1 + strlen(text))*sizeof(char));
    strcpy(ST[i].data,text);
    if (type == ID) {
        ST[i].value = 0;
        ST[i].init = 0;
    } else if (type == NUM) {
        ST[i].value = atoi(text);
        ST[i].init = 1;
    } else {
        fprintf(stderr, "Error: Unknown type %d for symbol table\n", type);
        --nsymb;
        return -1;
    }
    return i;
}

int setsymbol ( int i, int val )
{
    if ((i < 0) || (i >= nsymb)) {
        fprintf(stderr, "Error: Invalid index %d in symbol table\n", i);
        return -1;
    }
    if (ST[i].type != ID) {
        fprintf(stderr, "Error: Attempt to set non-ID in symbol table\n");
        return -1;
    }
    ST[i].value = val;
    ST[i].init = 1;
    return i;
}

int getsymbol ( char *id )
{
    int i;
    for (i=0; i<nsymb; ++i) {
        if (!strcmp(ST[i].data, id)) {
            if (ST[i].init == 0) {
                fprintf(stderr, "Error: Attempt to read uninitialized value\n");
                return 0;
            }
            return ST[i].value;
        }
    }
    fprintf(stderr, "Error: Invalid ID %s\n", id);
    return 0;
}

{
    if ((i < 0) || (i >= nsymb)) {
        fprintf(stderr, "Error: Unknown index %d in sylbol table\n", i);
        return 0;
    }
    if (ST[i].init == 0) {

{
    node *p;
    if ((i < 0) || (i >= nsymb)) {
        fprintf(stderr, "Invalid index %d in symbol table\n", i);
        return NULL;
    }
    p = (node *)malloc(sizeof(node));
    p -> type = ST[i].type;
    p -> data = i;

node *addnode ( int op, node *left, node *right )
{
    node *p;
    p = (node *)malloc(sizeof(node));
    p -> type = OP;
    p -> data = op;
    p -> L = left;
    p -> R = right;
    return p;
}

int intpow ( int a, int e )
{
    int prod = 1, a2pow = a;
    if (e < 0) return 0;
    while (e > 0) {
        if (e % 2) prod *= a2pow;
        e /= 2;
        a2pow *= a2pow;
    }
    return prod;
}

int evaltree ( node *p )
{
    int val, lval, rval;
    if (p == NULL) {
        fprintf(stderr, "Error: Empty expression tree\n");
        return 0;
    }
    if ( (p -> type == ID) || (p -> type == NUM) ) {
        val = getvalue(p -> data);
    } else if (p -> type == OP) {
        lval = evaltree(p -> L);
        rval = evaltree(p -> R);
        switch (p -> data) {
            case ADD: val = lval + rval; break;
            case SUB: val = lval - rval; break;
            case MUL: val = lval * rval; break;
            case DIV: val = lval / rval; break;
            case REM: val = lval % rval; break;
            case POW: val = intpow(lval,rval); break;
            default:
                fprintf(stderr, "Error: Unknown opcode %d\n", p -> data);
                val = 0;
                break;
        }
    } else {
        fprintf(stderr, "Error: Unknown node type in expression tree\n");
        val = 0;
    }
    free(p);
    return val;
}

// Function to insert an identifier into the symbol table

{
    stnode *p;

{

    p = (stacknode *)malloc(sizeof(stacknode));

    p = S;
    // Move the stack pointer to the next node 
    S = S->next;

    free(p);

    return S;

/*
The following parse() function implements the textbook 
predictive parsing algorithm under an LL(1) grammar.

The predictive parsing table contains the following non-error entries:
  M[EXPR,LP]   = EXPR -> ( OP ARG ARG )
  M[OP,binop]  = OP -> binop
  M[ARG,ID]    = ARG -> ID 
  M[ARG,NUM]   = ARG -> NUM 
  M[ARG,LP]    = ARG -> ( EXPR )
*/

// Function to perform predictive parsing and construct the parse tree 
parsetree parse(symbtbl *T, constbl *C)
{
    int A, a;               // A = stack symbol, a = current input token 
    stack S = NULL;         // Initialize parsing stack
    parsetree PT = NULL;    // Initialize parse tree
    treenode *p = NULL, *q; // Pointers used while creating the tree

    // Push the starting grammar symbol EXPR onto the stack 
    S = push(S, EXPR);
    // Get the first token from the lexical analyzer 
    a = yylex();

    // Continue parsing until the stack becomes empty 
    while (1) {
        // Stop when there are no more symbols in the stack 
        if (S == NULL)
            break;
        // Get and remove the top symbol from the stack 
        A = top(S);
        S = pop(S);

        // Handle the EXPR non-terminal 
        if (A == EXPR) {
            // EXPR must begin with a left parenthesis 
            if (a != LP) {
                fprintf(stderr, "*** Error: LP expected in place of %s\n", yytext); 
                exit(2);
            }
            // Push the right-hand side of EXPR -> ( OP ARG ARG ) onto the stack in reverse order 
            S = push(S, RP);
            S = push(S, ARG); 
            S = push(S, ARG);
            S = push(S, OP);
            S = push(S, LP);
        }
        // Handle the ARG non-terminal 
        else if (A == ARG) {
            // ARG can be an identifier 
            if (a == ID)
                S = push(S, ID);
            // ARG can be a number 
            else if (a == NUM)
                S = push(S, NUM);
            // ARG can be another expression 
            else if (a == LP)
                S = push(S, EXPR);
            // Report an error for an invalid argument 
            else {
                fprintf(stderr, "*** Error: ID/NUM/LP expected in place of %s\n", yytext); 
                exit(3);
            }
        }
        // Handle the OP non-terminal 
        else if (A == OP) {
            // Check whether the current token is a valid operator 
            if (!binop(a)) {
                fprintf(stderr, "*** Error: Invalid operator %s found\n", yytext); 
                exit(4);
            }
            // Push the operator token onto the stack
            S = push(S, a);
        }
        // Handle terminal symbols 
        else {
            // Handle an identifier token 
            if (A == ID) {
                // The current input token must be an identifier 
                if (a != ID) {
                    fprintf(stderr, "*** ERROR: ID expected in place of %s\n", yytext); 
                    exit(5);
                }
                // Check whether the current parent can have another child 
                if ((p == NULL) || ((p->L != NULL) && (p->R != NULL))) {
                    fprintf(stderr, "*** Error: Invalid id %s found\n", yytext); 
                    exit(6);
                }
                // Create the left or right child node 
                if (p->L == NULL)
                    q = p->L = (treenode *)malloc(sizeof(treenode)); 
                else
                    q = p->R = (treenode *)malloc(sizeof(treenode));
                // Mark the node as an identifier node 
                q->type = ID;
                // Insert the identifier into the symbol table
                *T = insertid(*T, yytext, &((q->ref).id));
                // Initialize child pointers 
                q->L = q->R = NULL;
                // Set the parent pointer 
                q->P = p;
            }
            // Handle a number token 
            else if (A == NUM) {
                // The current input token must be a number
                if (a != NUM) {
                    fprintf(stderr, "*** ERROR: Number expected in place of %s\n", yytext); 
                    exit(7);
                }
                // Check whether the current parent can have another child 
                if ((p == NULL) || ((p->L != NULL) && (p->R != NULL))) {
                    fprintf(stderr, "*** Error: Invalid number %s found\n", yytext); 
                    exit(8);
                }
                // Create the left or right child node 
                if (p->L == NULL)
                    q = p->L = (treenode *)malloc(sizeof(treenode)); 
                else
                    q = p->R = (treenode *)malloc(sizeof(treenode));
                // Mark the node as a number node 
                q->type = NUM;
                // Insert the number into the constant table
                *C = insertnum(*C, yytext, &((q->ref).num));
                // Initialize child pointers 
                q->L = q->R = NULL;
                // Set the parent pointer 
                q->P = p;
            }
            // Handle the left parenthesis 
            else if (A == LP) {
                // Check whether the current token is a left parenthesis 
                if (a != LP) {
                    fprintf(stderr, "*** Error: Left parenthesis expected in place of %s\n", yytext); 
                    exit(9);
                }
            }
            // Handle the right parenthesis 
            else if (A == RP) {
                // Check whether the current token is a right parenthesis 
                if (a != RP) {
                    fprintf(stderr, "*** Error: Right parenthesis expected in place of %s\n", yytext); 
                    exit(10);
                }
                // Move to the parent until a node without a right child is found or the root node is reached 
                while ((p != PT) && (p->R != NULL))
                    p = p->P;
            }
            // Handle binary operator tokens 
            else if (binop(A)) {
                // Create a new operator node 
                if (PT == NULL) {
                    // Create the root node of the parse tree
                    p = PT = (treenode *)malloc(sizeof(treenode));
                    // Root node has no parent 
                    PT->P = NULL;
                } else {
                    // Add the new operator as the left child if possible 
                    if (p->L == NULL)
                        q = p->L = (treenode *)malloc(sizeof(treenode));
                    // Otherwise add it as the right child 
                    else if (p->R == NULL)
                        q = p->R = (treenode *)malloc(sizeof(treenode));
                    // Report an error if both children already exist 
                    else {
                        fprintf(stderr, "*** Error: Cannot create subtree for operator %s (%d)\n", yytext, p->type); 
                        exit(11);
                    }
                    // Set the parent pointer 
                    q->P = p;
                    // Make the new operator node the current parent 
                    p = q;
                }
                // Mark the node as an operator node 
                p->type = OP;
                // Allocate memory for the operator string
                (p->ref).op = (char *)malloc((1 + strlen(yytext)) * sizeof(char));
                // Copy the operator into the tree node 
                strcpy((p->ref).op, yytext);
                // Initialize child pointers 
                p->L = p->R = NULL;
            }

            // The current input token has been consumed.
            // Read the next token if the stack is not empty. 
            if (S != NULL)
                a = yylex();
        }
    }

    // Check whether there are extra characters in the input 
    if (S != NULL) {
        fprintf(stderr, "*** Error: Extra characters in input\n"); 
        exit(12);
    }
    // Display successful parsing message 
    printf("Parsing is successful\n");
    // Return the constructed parse tree 
    return PT;
}

// Function to print the parse tree 
void prntree(parsetree PT, int level)
{
    int l;
    // Return if the tree is empty 
    if (PT == NULL)
        return;
    // Print indentation according to the tree level 
    for (l = 0; l < level - 1; ++l)
        printf("   ");
    // Print an arrow for non-root nodes 
    if (level)
        printf(" ---> ");
    // Print operator node 
    if (PT->type == OP)
        printf("OP(%s)", (PT->ref).op);
    // Print identifier node 
    else if (PT->type == ID)
        printf("ID(%s)", (PT->ref).id->id);
    // Print number node
    else if (PT->type == NUM)
        printf("NUM(%d)", (PT->ref).num->value); 
    printf("\n");
    // Recursively print the left subtree 
    prntree(PT->L, level + 1);
    // Recursively print the right subtree 
    prntree(PT->R, level + 1);
}

// Function to print all identifiers in the symbol table 
void prnsymbols(symbtbl T)
{
    // Traverse the symbol table 
    while (T) {
        // Print the identifier 
        printf("\"%s\"\n", T->id);
        // Move to the next symbol 
        T = T->next;
    }
}

// Function to read values for all identifiers
void readvalues(symbtbl T)
{
    int tok;
    // Stop if the symbol table is empty 
    if (T == NULL)
        return;
    // Recursively process the remaining symbols 
    readvalues(T->next);
    // Read the next token from the input 
    tok = yylex();
    // The value must be a number 
    if (tok != NUM) {
        fprintf(stderr, "*** Error: %s is not an integer value for %s\n", yytext, T->id); 
        exit(13);
    }
    // Convert the input string to an integer 
    T->value = atoi(yytext);
    // Display the identifier and its assigned value 
    printf("%s = %d\n", T->id, T->value);
}

// Function to evaluate the parse tree 
int evaltree(parsetree PT)
{
    // Return zero if the tree is empty 
    if (PT == NULL)
        return 0;
    // Evaluate according to the node type 
    switch (PT->type) {
        // Return the value stored for an identifier 
        case ID:
            return ((PT->ref).id)->value;
        // Return the value of a number 
        case NUM:
            return ((PT->ref).num)->value;
        // Evaluate an operator node 
        case OP:
            // Perform addition
            if (!strcmp((PT->ref).op, "+"))
                return evaltree(PT->L) + evaltree(PT->R);
            // Perform subtraction
            if (!strcmp((PT->ref).op, "-"))
                return evaltree(PT->L) - evaltree(PT->R);
            // Perform multiplication
            if (!strcmp((PT->ref).op, "*"))
                return evaltree(PT->L) * evaltree(PT->R);
            // Perform division
            if (!strcmp((PT->ref).op, "/"))
                return evaltree(PT->L) / evaltree(PT->R);
            // Perform remainder operation 
            if (!strcmp((PT->ref).op, "%"))
                return evaltree(PT->L) % evaltree(PT->R);
    }
    return 0;
}

  To implement predictive parsing for arithmetic expressions using an LL(1) 
  grammar, construct and analyze the corresponding parse tree, and use GDB

OVERVIEW & GRAMMAR:
  Predictive parsing is a top-down parsing technique used for LL(1) grammars. 
  In this experiment, arithmetic expressions are represented in parenthesized 
  list form (Lisp-like prefix notation).

  The grammar used is:
    EXPR -> ( OP ARG ARG )
    OP   -> + | - | * | / | %
    ARG  -> ID | NUM | EXPR

  - EXPR is the start symbol.
  - The grammar supports binary arithmetic operators, identifiers, integers, 
    and nested expressions.
  - Operator nodes form internal nodes; identifiers & numbers form leaf nodes.
  - After parsing, values are assigned to identifiers and the expression is 
    evaluated recursively by traversing the parse tree.

Predictive Parsing Table M(A, a):
  M[EXPR, LP]   = EXPR -> ( OP ARG ARG )
  M[OP, binop]  = OP -> binop
  M[ARG, ID]    = ARG -> ID
  M[ARG, NUM]   = ARG -> NUM
  M[ARG, LP]    = ARG -> ( EXPR )

Sample Input:
  (+ (* a b) d)
  With variable values: a = 2, b = 3, d = 4
  Result = (2 * 3) + 4 = 10

/* Global lexical analyzer state */
static char yytext[256];
static const char *input_cursor = NULL;

static int yylex(void) {
    while (*input_cursor && isspace((unsigned char)*input_cursor)) {
        input_cursor++;
    }

    if (*input_cursor == '\0') {
        yytext[0] = '\0';
        return END_OF_INPUT;
    }

    if (*input_cursor == '(') {
        strcpy(yytext, "(");
        input_cursor++;
        return LP;
    }
    if (*input_cursor == ')') {
        strcpy(yytext, ")");
        input_cursor++;
        return RP;
    }
    if (*input_cursor == '+') {
        strcpy(yytext, "+");
        input_cursor++;
        return ADD;
    }
    if (*input_cursor == '-' && !isdigit((unsigned char)*(input_cursor + 1))) {
        strcpy(yytext, "-");
        input_cursor++;
        return SUB;
    }
    if (*input_cursor == '*') {
        strcpy(yytext, "*");
        input_cursor++;
        return MUL;
    }
    if (*input_cursor == '/') {
        strcpy(yytext, "/");
        input_cursor++;
        return DIV;
    }
    if (*input_cursor == '%') {
        strcpy(yytext, "%");
        input_cursor++;
        return REM;
    }

    if (isdigit((unsigned char)*input_cursor) || (*input_cursor == '-' && isdigit((unsigned char)*(input_cursor + 1)))) {
        int i = 0;
        if (*input_cursor == '-') yytext[i++] = *input_cursor++;
        while (isdigit((unsigned char)*input_cursor)) {
            yytext[i++] = *input_cursor++;
        }
        yytext[i] = '\0';
        return NUM;
    }

    if (isalpha((unsigned char)*input_cursor) || *input_cursor == '_') {
        int i = 0;
        while (isalnum((unsigned char)*input_cursor) || *input_cursor == '_') {
            yytext[i++] = *input_cursor++;
        }
        yytext[i] = '\0';
        return ID;
    }

    yytext[0] = *input_cursor++;
    yytext[1] = '\0';
    return INV;
}

/* Symbol Table Functions */
symbtbl insertid(symbtbl T, char *id_str, stnode **ref) {
    stnode *p = T;
    while (p != NULL) {
        if (!strcmp(p->id, id_str)) {
            *ref = p;
            return T;
        }
        p = p->next;
    }
    p = (stnode *)malloc(sizeof(stnode));
    p->id = (char *)malloc((1 + strlen(id_str)) * sizeof(char));
    strcpy(p->id, id_str);
    p->value = 0;
    p->next = T;
    *ref = p;
    return p;

/* Stack Operations */
int top(stack S) {
    if (S == NULL) {
        fprintf(stderr, "*** Read from an empty stack\n");
        exit(1);
    }
    return S->item;
}

stack push(stack S, int item) {
    stacknode *p = (stacknode *)malloc(sizeof(stacknode));
    p->item = item;
    p->next = S;
    return p;
}

stack pop(stack S) {
    if (S == NULL) {
        fprintf(stderr, "*** Pop from an empty stack\n");
        exit(1);
    }
    stacknode *p = S;
    S = S->next;
    free(p);
    return S;
}

/* Predictive Parsing Algorithm */
parsetree parse(symbtbl *T, constbl *C) {
    int A, a;
    stack S = NULL;
    parsetree PT = NULL;
    treenode *p = NULL, *q;

    S = push(S, EXPR);
    a = yylex();

    while (1) {
        if (S == NULL) break;
        A = top(S);
        S = pop(S);

        if (A == EXPR) {
            if (a != LP) {
                fprintf(stderr, "*** Error: LP expected in place of %s\n", yytext);
                exit(2);
            }
            S = push(S, RP);
            S = push(S, ARG);
            S = push(S, ARG);
            S = push(S, OP);
            S = push(S, LP);
        } else if (A == ARG) {
            if (a == ID) {
                S = push(S, ID);
            } else if (a == NUM) {
                S = push(S, NUM);
            } else if (a == LP) {
                S = push(S, EXPR);
            } else {
                fprintf(stderr, "*** Error: ID/NUM/LP expected in place of %s\n", yytext);
                exit(3);
            }
        } else if (A == OP) {
            if (!binop(a)) {
                fprintf(stderr, "*** Error: Invalid operator %s found\n", yytext);
                exit(4);
            }
            S = push(S, a);
        } else {
            // Terminal symbols
            if (A == ID) {
                if (a != ID) {
                    fprintf(stderr, "*** ERROR: ID expected in place of %s\n", yytext);
                    exit(5);
                }
                if ((p == NULL) || ((p->L != NULL) && (p->R != NULL))) {
                    fprintf(stderr, "*** Error: Invalid id %s found\n", yytext);
                    exit(6);
                }
                if (p->L == NULL) q = p->L = (treenode *)malloc(sizeof(treenode));
                else              q = p->R = (treenode *)malloc(sizeof(treenode));

                q->type = ID;
                *T = insertid(*T, yytext, &((q->ref).id));
                q->L = q->R = NULL;
                q->P = p;
            } else if (A == NUM) {
                if (a != NUM) {
                    fprintf(stderr, "*** ERROR: Number expected in place of %s\n", yytext);

                if ((p == NULL) || ((p->L != NULL) && (p->R != NULL))) {

                if (p->L == NULL) q = p->L = (treenode *)malloc(sizeof(treenode));
                else              q = p->R = (treenode *)malloc(sizeof(treenode));

                q->type = NUM;
                *C = insertnum(*C, yytext, &((q->ref).num));
                q->L = q->R = NULL;
                q->P = p;
            } else if (A == LP) {
                if (a != LP) {
                    fprintf(stderr, "*** Error: Left parenthesis expected in place of %s\n", yytext);
                    exit(9);
                }
            } else if (A == RP) {
                if (a != RP) {
                    fprintf(stderr, "*** Error: Right parenthesis expected in place of %s\n", yytext);
                    exit(10);
                }
                while ((p != PT) && (p->R != NULL)) {
                    p = p->P;
                }
            } else if (binop(A)) {
                if (PT == NULL) {
                    p = PT = (treenode *)malloc(sizeof(treenode));
                    PT->P = NULL;
                } else {
                    if (p->L == NULL)      q = p->L = (treenode *)malloc(sizeof(treenode));
                    else if (p->R == NULL) q = p->R = (treenode *)malloc(sizeof(treenode));
                    else {
                        fprintf(stderr, "*** Error: Cannot create subtree for operator %s\n", yytext);
                        exit(11);
                    }
                    q->P = p;
                    p = q;
                }
                p->type = OP;
                (p->ref).op = (char *)malloc((1 + strlen(yytext)) * sizeof(char));
                strcpy((p->ref).op, yytext);
                p->L = p->R = NULL;
            }

            if (S != NULL) {
                a = yylex();
            }
        }
    }

    if (a != END_OF_INPUT && a != RP) {
        // If remaining tokens exist after empty stack
    }

    printf("Parsing is successful\n");
    return PT;
}

/* Evaluate parse tree */
int evaltree(parsetree PT) {
    if (PT == NULL) return 0;
    switch (PT->type) {
        case ID:
            return ((PT->ref).id)->value;
        case NUM:
            return ((PT->ref).num)->value;
        case OP:
            if (!strcmp((PT->ref).op, "+"))
                return evaltree(PT->L) + evaltree(PT->R);
            if (!strcmp((PT->ref).op, "-"))
                return evaltree(PT->L) - evaltree(PT->R);
            if (!strcmp((PT->ref).op, "*"))
                return evaltree(PT->L) * evaltree(PT->R);
            if (!strcmp((PT->ref).op, "/")) {
                int r = evaltree(PT->R);
                return (r != 0) ? (evaltree(PT->L) / r) : 0;
            }
            if (!strcmp((PT->ref).op, "%")) {
                int r = evaltree(PT->R);
                return (r != 0) ? (evaltree(PT->L) % r) : 0;
            }
    }
    return 0;
}

/*
================================================================================
EX NO: 9  23Z511 – SYSTEM SOFTWARE LABORATORY
PREDICTIVE PARSING TABLE & STACK SIMULATION
================================================================================
QUESTION (from Page 18 of Lab Manual):
  Write the predictive parsing table for a sample input.
  Sample Input: (+ (* a b) d)

GRAMMAR:
  EXPR -> ( OP ARG ARG )
  OP   -> binop
  ARG  -> ID
  ARG  -> NUM
  ARG  -> ( EXPR )

  where binop = + | - | * | / | %

PREDICTIVE PARSING TABLE M(A, a):
-----------------------------------------------------------------------------------
Non-Terminal |      (                 |    ID     |    NUM    | binop  |   )   |  $
-----------------------------------------------------------------------------------
EXPR         | EXPR -> ( OP ARG ARG ) |    ---    |    ---    |  ---   |  ---  | ---
OP           |          ---           |    ---    |    ---    | OP->b  |  ---  | ---
ARG          | ARG -> ( EXPR )        | ARG -> ID | ARG -> NUM|  ---   |  ---  | ---
-----------------------------------------------------------------------------------

PARSING TRACE (for (+ (* a b) d) $):
  Stack                       Input                         Production / Action
  ---------------------------------------------------------------------------------
  EXPR $                      ( + ( * a b ) d ) $           EXPR -> ( OP ARG ARG )
  ( OP ARG ARG ) $            ( + ( * a b ) d ) $           Match (
  OP ARG ARG ) $              + ( * a b ) d ) $             OP -> binop (+)
  + ARG ARG ) $               + ( * a b ) d ) $             Match +
  ARG ARG ) $                 ( * a b ) d ) $               ARG -> ( EXPR )
  ( EXPR ) ARG ) $            ( * a b ) d ) $               Match (
  EXPR ) ARG ) $              * a b ) d ) $                 EXPR -> ( OP ARG ARG )
  ... (Nested expression processing)
  ARG ) $                     d ) $                         ARG -> ID
  d ) $                       d ) $                         Match d
  ) $                         ) $                           Match )
  $                           $                             Accept

#include <stdio.h>

static Symbol *table = NULL;

Symbol *add_symbol(const char *name)
{
    Symbol *s = table;

    while (s != NULL) {
        if (strcmp(s->name, name) == 0)
            return s;
        s = s->next;
    }

    s = malloc(sizeof(Symbol));
    if (!s) {
        perror("malloc");
        exit(EXIT_FAILURE);
    }

    s->name = malloc(strlen(name) + 1);
    if (!s->name) {
        perror("malloc");
        exit(EXIT_FAILURE);
    }

    strcpy(s->name, name);
    s->value = 0;
    s->next = table;
    table = s;

    return s;
}

Node *make_leaf(NodeType type, const char *name, double value)
{
    Node *n = malloc(sizeof(Node));
    if (!n) {
        perror("malloc");
        exit(EXIT_FAILURE);
    }

    n->type = type;
    n->value = value;
    n->name = NULL;
    n->op = '\0';
    n->left = NULL;
    n->right = NULL;

    if (name) {
        n->name = malloc(strlen(name) + 1);
        if (!n->name) {
            perror("malloc");
            exit(EXIT_FAILURE);
        }
        strcpy(n->name, name);
    }

    return n;
}

Node *make_node(char op, Node *left, Node *right)
{
    Node *n = make_leaf(OP_NODE, NULL, 0);
    n->op = op;
    n->left = left;
    n->right = right;
    return n;
}

double eval_tree(Node *node)
{
    if (node->type == NUM_NODE)
        return node->value;

    if (node->type == ID_NODE) {
        Symbol *s = add_symbol(node->name);
        return s->value;
    }

    double a = eval_tree(node->left);
    double b = eval_tree(node->right);

    switch (node->op) {
        case '+': return a + b;
        case '-': return a - b;
        case '*': return a * b;
        case '/':
            if (b == 0) {
                fprintf(stderr, "Error: division by zero\n");
                return 0;
            }
            return a / b;
    }

    return 0;
}

void free_tree(Node *node)
{
    if (!node)
        return;

    free_tree(node->left);
    free_tree(node->right);
    free(node->name);
    free(node);
}

    printf("Enter assignments or expressions ending with ;\n");
    printf("Example: x = 10;  x + 5;\n");
    yyparse();
    return 0;

typedef struct Symbol {
    char *name;
    int offset;
    struct Symbol *next;
} Symbol;

static Symbol *symbols = NULL;
static int next_offset = 0;
static int temp_count = 0;
static FILE *out = NULL;

static void ensure_output(void)
{
    if (!out) {
        out = fopen("intcode.c", "w");
        if (!out) {
            perror("intcode.c");
            exit(EXIT_FAILURE);
        }
    }
}

int symbol_offset(const char *name)
{
    Symbol *s;

    for (s = symbols; s; s = s->next) {
        if (strcmp(s->name, name) == 0)
            return s->offset;
    }

    s = malloc(sizeof(Symbol));
    if (!s) exit(EXIT_FAILURE);

    s->name = malloc(strlen(name) + 1);
    if (!s->name) exit(EXIT_FAILURE);

    strcpy(s->name, name);
    s->offset = next_offset++;
    s->next = symbols;
    symbols = s;

    return s->offset;
}

int new_temp(void)
{
    char name[40];

    snprintf(name, sizeof(name), "$%d", ++temp_count);
    return symbol_offset(name);
}

void emit_program_start(void)
{
    ensure_output();

    fprintf(out,
        "#include <stdio.h>\n"
        "#include <stdlib.h>\n"
        "int MEM[1000];\n"
        "int main(void) {\n");
}

void emit_program_end(void)
{
    ensure_output();

    fprintf(out, "return 0;\n}\n");
    fclose(out);
    out = NULL;
}

    n->left = NULL;

    return n;

Node *make_binary(char op, Node *left, Node *right)
{
    Node *n = malloc(sizeof(Node));
    if (!n) exit(EXIT_FAILURE);

    n->type = OP_NODE;
    n->value = 0;
    n->op = op;
    n->left = left;
    n->right = right;
    n->inherited = 0;
    n->synthesized = 0;
    return n;
}

/* Compute inherited and synthesized attributes recursively. */
void setatt(Node *n)
{
    if (!n) return;

    if (n->type == NUMBER_NODE) {
        n->synthesized = n->value;
        return;
    }

    if (n->type == X_NODE) {
        n->synthesized = 0;
        return;
    }

    setatt(n->left);
    setatt(n->right);

    if (n->op == '+')
        n->synthesized =
            n->left->synthesized + n->right->synthesized;
    else if (n->op == '-')
        n->synthesized =
            n->left->synthesized - n->right->synthesized;
    else if (n->op == '*')
        n->synthesized =
            n->left->synthesized * n->right->synthesized;
    else if (n->op == '^') {
        int base = n->left->synthesized;
        int exponent = n->right->synthesized;
        int result = 1;

        while (exponent-- > 0)
            result *= base;

        n->synthesized = result;
    }
}

/* Print the annotated parse tree. */
void printtree(Node *n, int depth)
{
    if (!n) return;

    for (int i = 0; i < depth; i++)
        printf("  ");

    if (n->type == NUMBER_NODE)
        printf("NUM(%d), value=%d\n",
               n->value, n->synthesized);
    else if (n->type == X_NODE)
        printf("X, value=%d\n", n->synthesized);
    else
        printf("OP(%c), value=%d\n",
               n->op, n->synthesized);

    printtree(n->left, depth + 1);
    printtree(n->right, depth + 1);
}

/* Evaluate the polynomial for a given x. */
int evalpoly(Node *n, int x)
{
    if (!n) return 0;

    if (n->type == NUMBER_NODE)
        return n->value;

    if (n->type == X_NODE)
        return x;

    int a = evalpoly(n->left, x);
    int b = evalpoly(n->right, x);

    switch (n->op) {
        case '+': return a + b;
        case '-': return a - b;
        case '*': return a * b;

        case '^': {
            int result = 1;
            if (b < 0) return 0;

            while (b-- > 0)
                result *= a;

            return result;
        }
    }

    return 0;
}

/*
 * Print derivative terms in input order.
 * Supports constants, x, x^n, and coefficient*x^n.
 */
static void derivative_terms(Node *n, int sign, int *first)
{
    if (!n) return;

    if (n->type == OP_NODE &&
        (n->op == '+' || n->op == '-')) {
        derivative_terms(n->left, sign, first);
        derivative_terms(n->right,
                         n->op == '+' ? sign : -sign,
                         first);
        return;
    }

    int coefficient = 1;
    int exponent = 0;

    if (n->type == NUMBER_NODE) {
        return;
    } else if (n->type == X_NODE) {
        exponent = 1;
    } else if (n->type == OP_NODE && n->op == '*') {
        Node *a = n->left;
        Node *b = n->right;

        if (a->type == NUMBER_NODE) {
            coefficient = a->value;

            if (b->type == X_NODE)
                exponent = 1;
            else if (b->type == OP_NODE && b->op == '^' &&
                     b->left->type == X_NODE &&
                     b->right->type == NUMBER_NODE)
                exponent = b->right->value;
            else
                return;
        } else {
            return;
        }
    } else if (n->type == OP_NODE && n->op == '^' &&
               n->left->type == X_NODE &&
               n->right->type == NUMBER_NODE) {
        exponent = n->right->value;
    } else {
        return;
    }

    if (exponent == 0) return;

    int c = coefficient * exponent;

    if (!*first)
        printf(sign < 0 ? " - " : " + ");
    else if (sign < 0)
        printf("-");

    if (c < 0) {
        printf("%d", -c);
    } else {
        printf("%d", c);
    }

    if (exponent - 1 == 1)
        printf("*x");
    else if (exponent - 1 > 1)
        printf("*x^%d", exponent - 1);

    *first = 0;
}

void printderivative(Node *n)
{
    int first = 1;

    derivative_terms(n, 1, &first);

    if (first)
        printf("0");

    printf("\n");
}

void free_tree(Node *n)
{
    if (!n) return;

    free_tree(n->left);
    free_tree(n->right);
    free(n);
}

int main(void)
{
    if (yyparse() != 0 || root == NULL) {
        fprintf(stderr, "Invalid polynomial\n");
        return 1;
    }

    setatt(root);

    printf("Annotated parse tree:\n");
    printtree(root, 0);

    printf("\nPolynomial values:\n");

    for (int x = -5; x <= 5; x++)
        printf("x = %d, value = %d\n",
               x, evalpoly(root, x));

    printf("\nDerivative: ");
    printderivative(root);

    free_tree(root);
    return 0;
}
